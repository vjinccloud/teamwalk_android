package com.taiwanlife.teamwalk.code_view_only;

import java.awt.image.BufferedImage;

import javax.imageio.ImageIO;
import javax.servlet.ServletOutputStream;
import javax.servlet.http.HttpServletResponse;

import org.apache.commons.lang.StringUtils;
import org.apache.log4j.Logger;
import org.owasp.esapi.ESAPI;

import com.twlife.csso.action.AbstractAction;
import com.twlife.csso.servlet.CssoHttpServletResponse;
import com.twlife.csso.util.Constants;

public class CodeAction extends AbstractAction {

	private static final long serialVersionUID = -3788235212841379369L;

	private static final Logger logger = Logger.getLogger(CodeAction.class);

	/**
	 * 
	 */
	private String code;

	/**
	 * 驗證碼圖片的寬度。
	 */
	private int width;

	/**
	 * 驗證碼圖片的高度。
	 */
	private int height;

	/**
	 * 驗證碼字符個數
	 */
	private int codeCount;

	/**
	 * 
	 */
	private char[] codeSequence = new char[] { 'a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'k', 'm', 'n', 'p', 'q', 'r', 's', 't', 'u', 'v', 'w', 'x',
			'y', 'z', '2', '3', '4', '5', '6', '7', '8', '9' };

	public void prepare() throws Exception {
		super.prepare();
	}

	public String code() throws Exception {

		StringBuffer randomCode = new StringBuffer();
		for (int i = 0; i < codeCount; i++) {
			String strRand = String.valueOf(codeSequence[ESAPI.randomizer().getRandomInteger(1, codeSequence.length)]);
			randomCode.append(strRand);
		}

		// 禁止圖像緩存。
		HttpServletResponse resp = getResponse();
		resp.setHeader("Pragma", "no-cache");
		resp.setHeader("Cache-Control", "no-cache");
		resp.setDateHeader("Expires", 0);
		resp.setContentType("image/jpeg");

		// 將圖像輸出到Servlet輸出流中。
		TcavWordRenderer rd = new TcavWordRenderer();
		BufferedImage buffImg = rd.renderWord(randomCode.toString(), width, height);

		// 將四位數字的驗證碼保存到Session中。
		getRequest().getSession().setAttribute(Constants.VALIDATE_CODE, randomCode.toString());

		ServletOutputStream sos = null;
		try {
			sos = resp.getOutputStream();
			ImageIO.write(buffImg, "jpeg", sos);
		} catch (Exception e) {
			logger.error(e);
		} finally {
			if (null != sos) {
				sos.close();
			}
		}
		return null;
	}

	/**
	 * 驗證驗證碼 [USER 要求不分大小寫]
	 * 
	 * @return
	 */
	public String verifyCode() throws Exception {
		String code = "";
		CssoHttpServletResponse csr = new CssoHttpServletResponse(getResponse());
		try {
			if (null != getRequest().getSession().getAttribute(Constants.VALIDATE_CODE)) {
				String validateC = (String) getRequest().getSession().getAttribute(Constants.VALIDATE_CODE);
				if (null == getCode() || "".equals(getCode())) {
					code = "ERROR01";
				} else {
					if(StringUtils.equalsIgnoreCase(validateC, getCode())){
						code = "OK";
					} else {
						code = "ERROR02";
					}
				}
			} else {
				code = "ERROR03";
			}
		} catch (Exception e) {
			logger.error(e);
			code = "ERROR04";
		} finally {
			csr.getCssoWriter().print(code);
			csr.getCssoWriter().flush();
		}
		return null;
	}

	public int getCodeCount() {
		return codeCount;
	}

	public void setCodeCount(int codeCount) {
		this.codeCount = codeCount;
	}

	public String getCode() {
		return code;
	}

	public void setCode(String code) {
		this.code = code;
	}

	public int getWidth() {
		return width;
	}

	public void setWidth(int width) {
		this.width = width;
	}

	public int getHeight() {
		return height;
	}

	public void setHeight(int height) {
		this.height = height;
	}

}
