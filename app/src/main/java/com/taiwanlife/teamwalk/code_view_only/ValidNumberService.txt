package com.taiwanlife.teamwalk.code_view_only;

import java.awt.image.BufferedImage;
import java.io.IOException;
import java.security.SecureRandom;

import javax.imageio.ImageIO;
import jakarta.servlet.http.HttpServletResponse;
import jakarta.servlet.http.HttpSession;

import org.apache.commons.lang3.StringUtils;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import com.google.code.kaptcha.Producer;
import com.taiwanlife.exception.CheckVaildCodeException;
import com.taiwanlife.exception.CheckVaildCodeNotMatchException;

import lombok.extern.slf4j.Slf4j;

@Service
@Slf4j
public class ValidNumberService {

	private final static String VALID_NUMBER_SESSION_ATTR = "valid_number";
	
	@Autowired
	private HttpSession session;
	
	@Autowired
    private Producer captchaProducer;
	
	public String getValidNumber(String category) {

		log.trace("[getValidNumber] session id:" + session.getId());

		if (session != null && session.getAttribute(getSessionAttrName(category)) != null) {

			return String.valueOf(session.getAttribute(getSessionAttrName(category)));

		}

		return null;

	}
	
	public void createImageCode(String count, String type, HttpServletResponse response, String category) throws IOException {
		// 設定Response型態 
        response.setContentType("image/jpeg");
        response.setHeader("Pragma", "No-cache");
        response.setHeader("Cache-Control", "no-cache");
        response.setDateHeader("Expires", 0);
        
        // ***** 【產生隨機驗證碼 】 ***** //
        char[] simpleCode = {'1','2','3','4','5','6','7','8','9','0'};
		
		char[] difficultCode = { '1','2','3','4','5','6','7', '8','9','0',
				 'a','b','c','d','e','f','g','h','i','j','k','l','m','n','o','p','q','r','s','t','u','v','w','x','y','z'};
			        
		//filter 數字0、1，英文l、o、i
		difficultCode = new String(difficultCode).replaceAll("[01oil]", "").toCharArray();
		
		Integer validNumberCount = 6;
		String validType = "difficult";// difficult | simple

		if (StringUtils.isNotEmpty(count) && StringUtils.isNumeric(count))  validNumberCount = Integer.parseInt(count);
		if (StringUtils.isNotEmpty(type))  validType = type;	
		
		String sRand = "";
		char[] code = "simple".equalsIgnoreCase(validType) ? simpleCode : difficultCode;
		
		try {
			SecureRandom random = SecureRandom.getInstance("SHA1PRNG");
			for (int i = 0; i < validNumberCount; i++) {
		          String rand = String.valueOf(code[random.nextInt(code.length)]);
		          sRand += rand;
			}
		}catch(Exception e) {
			log.error("SecureRandom fail!! {}", e);
		}
		
		log.info("驗證碼：{}", sRand);
		session.setAttribute(getSessionAttrName(category), sRand);
		
        // ***** 【產生隨機圖形驗證碼 】 ***** //
        BufferedImage image = captchaProducer.createImage(sRand);
        session.setAttribute(getSessionAttrName(category), sRand);
        
		ImageIO.write(image, "jpeg", response.getOutputStream());
		
		response.getOutputStream().flush();
		response.getOutputStream().close();
        return;
	}
	
	public void checkValidCode(String inputValue, String category) {

		String validNumber = getValidNumber(category);

		if (StringUtils.isEmpty(validNumber)) {

			throw new CheckVaildCodeException();

		}

		if (!validNumber.equalsIgnoreCase(inputValue)) {

			throw new CheckVaildCodeNotMatchException(inputValue);

		}
		
		session.removeAttribute(getSessionAttrName(category));

	}

	private String getSessionAttrName(String category) {

		return StringUtils.isNotEmpty(category) ? VALID_NUMBER_SESSION_ATTR + "_" + category
				: VALID_NUMBER_SESSION_ATTR;

	}
}
